class Solution {
    public int[] plusOne(int[] digits) {
    for(int i=digits.length-1;i>=0;i--){
    if(digits[i]<9){
        digits[i]++;
        return digits;
    }
    else{
        digits[i]=0;
    }
    }
    // for single digit 9 mattum irundha
    int arr[]=new int[digits.length+1];
    arr[0]=1;
    return arr;
    
}
}
